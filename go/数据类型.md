## Go 语言中的基本数据类型

### 数字类型

```go
// 无符号整型
var u8 uint8 = 1    // 取值范围 0 - 255
var u16 uint16 = 1  // 取值范围 0 - 65535
var u32 uint32 = 1  // 取值范围 0 - 4294967295
var u64 uint64 = 1  // 取值范围 0 - 18446744073709551615

// 有符号整型
var i8 int8 = 1     // 取值范围 -128 - 127
var i16 int16 = 1   // 取值范围 -32768 - 32767
var i32 int32 = 1   // 取值范围 -2147483648 - 2147483647
var i64 int64 = 1   // 取值范围 -9223372036854775808 - 9223372036854775807

// 浮点型
var f32 float32 = 1.5   // 32 位浮点数，约 6 位十进制精度
var f64 float64 = 1.5   // 64 位浮点数，约 15 位十进制精度

// 复数
var c64 complex64 = 1 + 2i     // 由 float32 组成
var c128 complex128 = 1 + 2i   // 由 float64 组成

// 平台相关（取决于操作系统位数）
var u uint = 1      // 32 位系统为 uint32，64 位系统为 uint64
var i int = 1       // 32 位系统为 int32，64 位系统为 int64
var p uintptr = 1   // 足以保存指针的整数类型
```

### 字符类型

```go
// byte 是 uint8 的别名，用于表示 ASCII 字符（单字节）
var b byte = 'A'            // 等价于 var b byte = 65

// rune 是 int32 的别名，用于表示 Unicode 字符（多字节，如中文）
var r rune = '中'           // 等价于 var r rune = 20013

// 字符串本质上是字节序列，len() 返回的是字节数而非字符数
var s string = "A中"
fmt.Println(len(s))         // 输出 4（A 占 1 字节，中占 3 字节）
fmt.Println(len([]rune(s))) // 输出 2（按字符计数）
```

### 布尔类型

```go
// 布尔类型只有两个值：true 和 false
var flag bool = true
var done bool = false

// 声明时不给初始值，默认为 false
var zero bool

// 不能与数字类型相互转换，也不能用于算术运算
// var n int = 1
// var b bool = n          // 编译错误：不能将 int 转换为 bool
// fmt.Println(flag + 1)   // 编译错误：无效的运算

// 常用于条件判断
if flag {
    fmt.Println("flag is true")
}
```

## 复合类型

### 数组

```go
// 数组是固定长度的同类型元素序列，长度是类型的一部分
var arr [5]int              // 长度为 5 的 int 数组，元素默认为 0
var arr2 [3]string = [3]string{"a", "b", "c"}

// 声明时直接初始化，长度可省略（用 ... 让编译器推断）
var arr3 = [...]int{1, 2, 3}   // 长度为 3

// 索引从 0 开始
fmt.Println(arr2[0])           // 输出 a
arr[0] = 10                    // 修改元素
fmt.Println(len(arr))          // 输出 5，len() 获取数组长度

// 数组是值类型，赋值或传参会复制整个数组
arr4 := arr3
arr4[0] = 100
fmt.Println(arr3[0])           // 输出 1，arr3 不受影响

// 数组长度固定，不能动态增删
// arr[5] = 10  // 编译错误：索引越界
```

### 切片

```go
// 切片是数组的引用，长度可动态变化
var s []int                 // 声明一个空切片，默认值为 nil
s2 := []int{1, 2, 3}        // 声明并初始化切片
s3 := make([]int, 3)        // 用 make 创建长度为 3 的切片，元素默认为 0
s4 := make([]int, 3, 5)     // 长度为 3，容量为 5

// 对数组切片，得到切片（左闭右开）
arr := [5]int{1, 2, 3, 4, 5}
sl := arr[1:4]              // {2, 3, 4}，不包括索引 4

// 切片是引用类型，修改会影响底层数组
sl[0] = 99
fmt.Println(arr[1])         // 输出 99

// 追加元素，超出容量时自动扩容
s5 := []int{1, 2, 3}
s5 = append(s5, 4)          // 变为 {1, 2, 3, 4}
fmt.Println(len(s5), cap(s5))   // 输出长度和容量

// len() 获取长度，cap() 获取容量
fmt.Println(len(sl), cap(sl))   // 输出 3, 4
```

### Map

```go
// map 是键值对的集合，键必须是可比较类型（如 string、int）
var m map[string]int                // 声明一个 nil map，不能直接写入
m2 := map[string]int{"a": 1, "b": 2}  // 声明并初始化
m3 := make(map[string]int)          // 用 make 创建空 map

// 写入和读取
m3["a"] = 1
m3["a"] = 2                        // 更新已有键的值
fmt.Println(m3["a"])               // 输出 2

// 判断键是否存在：读取不存在的键返回零值，用 ok 判断
v, ok := m3["b"]
fmt.Println(v, ok)                 // 输出 0 false

// 删除键
delete(m3, "a")

// 遍历 map（顺序不固定）
for k, v := range m2 {
    fmt.Println(k, v)
}

// 获取元素个数
fmt.Println(len(m2))               // 输出 2

// map 是引用类型，直接赋值或传参共享底层数据
m4 := m2
m4["c"] = 3
fmt.Println(m2["c"])               // 输出 3
```
